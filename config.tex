\newcommand{\fullradialgradient}{%

  % Color de base
  \fill[mateBlueOuter]
    (current page.south west)
    rectangle
    (current page.north east);

  % Limitamos el dibujo a la diapositiva
  \clip
    (current page.south west)
    rectangle
    (current page.north east);

  % Degradado radial
  \shade[
    inner color=mateBlueInner,
    outer color=mateBlueOuter
  ]
    (current page.center)
    circle[radius=.72\paperwidth];
}

% Símbolos de la portada original de PowerPoint.
% Centros relativos a la página; tamaños relativos al ancho para evitar
% deformaciones. El blanco al 25 % deja visible el degradado bajo las figuras.
% Geometría compartida por los símbolos de portada y cabecera.
\newcommand{\mathsymbolshape}[1]{%
    \ifcase#1
      % Anillo: diámetro exterior 1, hueco de diámetro 0.38.
      \path[fill=white,fill opacity=.25,even odd rule]
        (0,0) circle[radius=.5] (0,0) circle[radius=.19];
    \or
      % Cuadrado hueco.
      \path[fill=white,fill opacity=.25,even odd rule]
        (-.5,-.5) rectangle (.5,.5)
        (-.21,-.21) rectangle (.21,.21);
    \or
      % Cruz de brazos rectos (también usada como signo de multiplicación).
      \path[fill=white,fill opacity=.25]
        (-.16,.5) -- (.16,.5) -- (.16,.16) -- (.5,.16) --
        (.5,-.16) -- (.16,-.16) -- (.16,-.5) -- (-.16,-.5) --
        (-.16,-.16) -- (-.5,-.16) -- (-.5,.16) -- (-.16,.16) -- cycle;
    \fi
}

\newcommand{\normalbackground}{%
\begin{tikzpicture}[remember picture,overlay]

  % Fondo blanco
  \fill[white]
    (current page.south west)
    rectangle
    (current page.north east);

  % ---------------------------------------------------------------------------
  % Cabecera azul degradada
  % ---------------------------------------------------------------------------

  \begin{scope}

    % Limitamos el degradado a la franja superior
    \clip
      (current page.north west)
      rectangle
      ([yshift=-0.95cm]current page.north east);

    % Fondo base azul
    \fill[mateBlueOuter]
      (current page.north west)
      rectangle
      ([yshift=-0.95cm]current page.north east);

    % Degradado radial
    \shade[
      inner color=mateBlueInner,
      outer color=mateBlueOuter
    ]
      ([yshift=-0.47cm]current page.north)
      circle[radius=.70\paperwidth];

    \headermathsymbols

  \end{scope}

\end{tikzpicture}
}

%
% Se genera automáticamente al escribir:
%
%   \section{Cálculo y estimación de errores e incertidumbre}
%
% =============================================================================

\AtBeginSection[]{
  \begingroup

  \usebackgroundtemplate{\topicbackground}

  \begin{frame}[plain]

    \begin{tikzpicture}[remember picture,overlay]

      % Número del tema
      \node[
        text=mateBlueDark,
        font=\sffamily\bfseries\fontsize{34}{38}\selectfont
      ]
      at ([yshift=1.05cm]current page.center)
      {Tema \insertsectionnumber};

      % Nombre del tema
      \node[
        text=mateText,
        align=center, text badly centered,
        text width=.82\paperwidth,
        font=\sffamily\bfseries\fontsize{23}{28}\selectfont
      ]
      at ([yshift=-.80cm]current page.center)
      {\insertsectionhead};

      \node[
        text=mateBlueDark, align=center,
        font=\sffamily\fontsize{10}{13}\selectfont
      ] at ([yshift=1.05cm]current page.south)
      {Matemáticas 2 \\ Grado en Ingeniería Informática};

    \end{tikzpicture}

  \end{frame}

  \endgroup
}

% =============================================================================
% SUBSECCIÓN / PORTADA DE SUBAPARTADO
%
% Se genera automáticamente al escribir:
%
%   \subsection{Del recorrido real al modelo matemático}
%
% =============================================================================

\AtBeginSubsection[]{
  \begingroup

  % Fondo totalmente blanco:
  % no queremos aquí la barra azul de los frames normales.
  \usebackgroundtemplate{}
  \setbeamercolor{background canvas}{bg=white}

  \begin{frame}[plain]

    \centering
    \vfill

    % Tema actual
    {\large
      \color{mateBlueDark}
      Tema \insertsectionnumber.
      \insertsectionhead
      \par
    }

    \vspace{1.2em}

    % Nombre de la subsección
    {\LARGE\bfseries
      \color{mateText}
      \insertsubsectionhead
      \par
    }

    \vspace{1em}

    {\color{mateBlueDark}
      \rule{0.55\textwidth}{1pt}
    }

    \vfill

  \end{frame}

  \endgroup
}
